### 算法细节（建议实现）
```ts
const IMPORTANCE_WEIGHTS = { vital: 5, important: 4, moderate: 3, minor: 2, trivial: 1 };

export function calculatePriority(
  importance: ImportanceLevel,
  dueDate: Date | null,
  currentTime: Date,
): number {
  const weight = IMPORTANCE_WEIGHTS[importance];
  if (weight === undefined) throw new PriorityCalculationError('Invalid importance');
  if (!(currentTime instanceof Date) || Number.isNaN(currentTime.getTime())) {
    throw new PriorityCalculationError('Invalid currentTime');
  }

  const isBacklog = !dueDate;
  const isInvalidDate = dueDate && Number.isNaN(dueDate.getTime());
  if (isInvalidDate) throw new PriorityCalculationError('Invalid dueDate');

  let overdueBoost = 0;
  let days = 999999; // backlog default
  if (!isBacklog) {
    const ms = dueDate!.getTime() - currentTime.getTime();
    if (ms < 0) overdueBoost = 50;
    days = Math.max(ms / 86400000, 0.01); // avoid div/0
  }

  const importanceScore = weight * 0.6 * 20; // map 1-5 → 12-60 range
  const timeScore = (1 / days) * 0.4 * 100;  // smaller days => bigger score, upper bounded by clamp
  const raw = importanceScore + timeScore + overdueBoost;
  return Math.max(0, Math.min(100, raw));
}
```
- 重要性映射乘以 20：保持满分 100 的量级，方便与 +50 overdue boost 协同。
- 时间项在 days 很小时会放大，最终由 clamp 控制。
